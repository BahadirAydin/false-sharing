cmake_minimum_required(VERSION 3.25)
project(cpp_work VERSION 0.1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(CPP_WORK_CACHE_LINE_SIZE 64 CACHE STRING "Cache-line separation in bytes; verify on the benchmark CPU")
set(CPP_WORK_SANITIZER "none" CACHE STRING "none, address, or thread")
set_property(CACHE CPP_WORK_SANITIZER PROPERTY STRINGS none address thread)
find_package(Threads REQUIRED)

add_library(cpp_work INTERFACE)
target_include_directories(cpp_work INTERFACE include)
target_compile_definitions(cpp_work INTERFACE CPP_WORK_CACHE_LINE_SIZE=${CPP_WORK_CACHE_LINE_SIZE})
target_compile_options(cpp_work INTERFACE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
target_link_libraries(cpp_work INTERFACE Threads::Threads)
if(CPP_WORK_SANITIZER STREQUAL "address")
  target_compile_options(cpp_work INTERFACE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(cpp_work INTERFACE -fsanitize=address,undefined)
elseif(CPP_WORK_SANITIZER STREQUAL "thread")
  target_compile_options(cpp_work INTERFACE -fsanitize=thread -fno-omit-frame-pointer)
  target_link_options(cpp_work INTERFACE -fsanitize=thread)
elseif(NOT CPP_WORK_SANITIZER STREQUAL "none")
  message(FATAL_ERROR "Unknown CPP_WORK_SANITIZER=${CPP_WORK_SANITIZER}")
endif()

include(CTest)
if(BUILD_TESTING)
  add_executable(spsc_queue_test tests/spsc_queue_test.cpp)
  target_link_libraries(spsc_queue_test PRIVATE cpp_work)
  add_test(NAME spsc_queue COMMAND spsc_queue_test)
  set_tests_properties(spsc_queue PROPERTIES TIMEOUT 60)
endif()
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  add_executable(false_sharing_bench benchmarks/false_sharing.cpp)
  target_link_libraries(false_sharing_bench PRIVATE cpp_work)
endif()
