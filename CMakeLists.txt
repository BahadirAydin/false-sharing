cmake_minimum_required(VERSION 3.25)
project(false_sharing VERSION 0.1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(FALSE_SHARING_CACHE_LINE_SIZE 64 CACHE STRING "Cache-line separation in bytes; verify on the benchmark CPU")
set(FALSE_SHARING_SANITIZER "none" CACHE STRING "none, address, or thread")
set_property(CACHE FALSE_SHARING_SANITIZER PROPERTY STRINGS none address thread)
find_package(Threads REQUIRED)

add_library(false_sharing INTERFACE)
target_include_directories(false_sharing INTERFACE include)
target_compile_definitions(false_sharing INTERFACE FALSE_SHARING_CACHE_LINE_SIZE=${FALSE_SHARING_CACHE_LINE_SIZE})
target_compile_options(false_sharing INTERFACE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
target_link_libraries(false_sharing INTERFACE Threads::Threads)
if(FALSE_SHARING_SANITIZER STREQUAL "address")
  target_compile_options(false_sharing INTERFACE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(false_sharing INTERFACE -fsanitize=address,undefined)
elseif(FALSE_SHARING_SANITIZER STREQUAL "thread")
  target_compile_options(false_sharing INTERFACE -fsanitize=thread -fno-omit-frame-pointer)
  target_link_options(false_sharing INTERFACE -fsanitize=thread)
elseif(NOT FALSE_SHARING_SANITIZER STREQUAL "none")
  message(FATAL_ERROR "Unknown FALSE_SHARING_SANITIZER=${FALSE_SHARING_SANITIZER}")
endif()

include(CTest)
if(BUILD_TESTING)
  add_executable(spsc_queue_test tests/spsc_queue_test.cpp)
  target_link_libraries(spsc_queue_test PRIVATE false_sharing)
  add_test(NAME spsc_queue COMMAND spsc_queue_test)
  set_tests_properties(spsc_queue PROPERTIES TIMEOUT 60)
endif()
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  add_executable(false_sharing_bench benchmarks/false_sharing.cpp)
  target_link_libraries(false_sharing_bench PRIVATE false_sharing)
endif()
